05 • ARDUINO IDE

ตัวอย่างโค้ด Arduino

ตัวอย่างนี้กำหนดให้พัดลมทำงานเมื่ออุณหภูมิสูงกว่า 30°C และหยุดเมื่ออุณหภูมิลดต่ำกว่า 28°C ส่วนปั๊มทำงานเมื่อความชื้นต่ำกว่า 80% และหยุดเมื่อสูงกว่า 88% ใช้ hysteresis เพื่อลดการสั่งเปิด–ปิดถี่

#include <DHT.h>

#define DHTPIN 2
#define DHTTYPE DHT22
#define FAN_RELAY 7
#define MIST_RELAY 8

// รีเลย์หลายรุ่นเป็น Active LOW
const int RELAY_ON = LOW;
const int RELAY_OFF = HIGH;

DHT dht(DHTPIN, DHTTYPE);
bool fanOn = false;
bool mistOn = false;

void setup() {
  Serial.begin(9600);
  pinMode(FAN_RELAY, OUTPUT);
  pinMode(MIST_RELAY, OUTPUT);
  digitalWrite(FAN_RELAY, RELAY_OFF);
  digitalWrite(MIST_RELAY, RELAY_OFF);
  dht.begin();
}

void loop() {
  float humidity = dht.readHumidity();
  float temperature = dht.readTemperature();

  if (isnan(humidity) || isnan(temperature)) {
    Serial.println("อ่านค่า DHT22 ไม่สำเร็จ");
    delay(2000);
    return;
  }

  // ควบคุมพัดลมด้วยช่วงเปิด-ปิด
  if (temperature >= 30.0) fanOn = true;
  if (temperature <= 28.0) fanOn = false;

  // ควบคุมระบบพ่นหมอกด้วยช่วงเปิด-ปิด
  if (humidity <= 80.0) mistOn = true;
  if (humidity >= 88.0) mistOn = false;

  digitalWrite(FAN_RELAY, fanOn ? RELAY_ON : RELAY_OFF);
  digitalWrite(MIST_RELAY, mistOn ? RELAY_ON : RELAY_OFF);

  Serial.print("Temperature: ");
  Serial.print(temperature);
  Serial.print(" C | Humidity: ");
  Serial.print(humidity);
  Serial.print(" % | Fan: ");
  Serial.print(fanOn ? "ON" : "OFF");
  Serial.print(" | Mist: ");
  Serial.println(mistOn ? "ON" : "OFF");

  delay(2000);
}
ก่อนใช้งาน: ตรวจสอบว่ารีเลย์ของคุณเป็น Active LOW หรือ Active HIGH หากเป็น Active HIGH ให้สลับค่า RELAY_ON และ RELAY_OFF และปรับเกณฑ์ให้เหมาะกับชนิดเห็ดและการทดลองจริง โค้ดนี้เป็นตัวอย่างควบคุมพื้นฐาน ไม่ใช่ระบบป้องกันความเสียหาย
06 • TEST & EVALUATION

วิธีทดสอบและประเมินผล

รายการทดสอบวิธีทดสอบผลที่คาดหวัง
เซนเซอร์ DHT22เปรียบเทียบค่ากับเครื่องวัดอ้างอิงอ่านค่าได้ต่อเนื่องและใกล้เคียงเครื่องอ้างอิง
พัดลมเพิ่มอุณหภูมิบริเวณเซนเซอร์อย่างปลอดภัยรีเลย์สั่งพัดลมเมื่อถึงเกณฑ์ที่ตั้งไว้
ระบบพ่นหมอกลดความชื้นจำลองหรือทดสอบตามเงื่อนไขปั๊มหรือวาล์วทำงานและหยุดตามเกณฑ์
พลังงานแสงอาทิตย์ตรวจแรงดันและสถานะชาร์จตามคู่มือแบตเตอรี่ได้รับการชาร์จอย่างเหมาะสม
ทดสอบต่อเนื่องบันทึกอุณหภูมิ ความชื้น และเวลาทำงานระบบทำงานเสถียร ไม่มีน้ำเข้ากล่องหรือสายร้อนผิดปกติ
